```python
class Solution(object):
    def fourSum(self, nums, target):
        """
        :type nums: List[int]
        :type target: int
        :rtype: List[List[int]]
        """
        nums.sort()
        n = len(nums) - 1
        result = []
        for i in range(n - 2):
            for j in range(i + 1, n - 1):
                k = j + 1
                p = n
                while k < p:
                    curr = nums[i] + nums[j] + nums[k] + nums[p]
                    if curr == target:
                        add = [nums[i], nums[j], nums[k], nums[p]]
                        if add not in result:
                            result.append(add)
                        p -= 1
                        k += 1
                    elif curr > target:
                        p -= 1
                    else:
                        k += 1

        return result
```

```python
                while l < r:
                    curr = nums[l] + nums[r]
                    if target == curr:
                        results.append( result + [nums[l], nums[r]])

                        r -= 1
```

## Ideal Solution
Recursion can be a very good option. Its runtime can be much faster.

Also, noticed the justification `if r-l+1 < N or N < 2 or target < nums[l]*N or target > nums[r]*N:  # early termination`. With this an early termination can be done, which save a lot of time.

```python
class Solution(object):
    def fourSum(self, nums, target):
        def findNsum(l, r, target, N, result, results):
            if r-l+1 < N or N < 2 or target < nums[l]*N or target > nums[r]*N:  # early termination
                return
            if N == 2: # two pointers solve sorted 2-sum problem
                while l < r:
                    s = nums[l] + nums[r]
                    if s == target:
                        results.append(result + [nums[l], nums[r]])
                        l += 1
                        while l < r and nums[l] == nums[l-1]:
                            l += 1
                    elif s < target:
                        l += 1
                    else:
                        r -= 1
            else: # recursively reduce N
                for i in range(l, r+1):
                    if i == l or (i > l and nums[i-1] != nums[i]):
                        findNsum(i+1, r, target-nums[i], N-1, result+[nums[i]], results)

        nums.sort()
        results = []
        findNsum(0, len(nums)-1, target, 4, [], results)
        return results
```

```python
class Solution(object):
    def fourSum(self, nums, target):
        """
        :type nums: List[int]
        :type target: int
        :rtype: List[List[int]]
        """
        nums.sort()
        n = len(nums)
        result = []
        
        if n < 4:
            return result
        
        for i in range(n - 3):
            # only do once for every i value
            if i > 0 and nums[i] == nums[i-1]:
                continue
            
            # if target is greater than the largest num * 4, skip
            if nums[i] + nums[i+1] + nums[i+2] + nums[i+3] > target:
                break
            # if target is smaller than the smallest num * 4, skip
            if nums[i] + nums[n-3] + nums[n-2] + nums[n-1] < target:
                continue
            
            for j in range(i + 1, n - 2):
                # only do once for every j value
                if j > i + 1 and nums[j] == nums[j-1]:
                    continue
                
                remaining = target - nums[i] - nums[j]
                
                # if remaining is greater than the largest num * 4, skip
                if remaining < nums[j+1] + nums[j+2]:
                    break
                # if remaining is smaller than the smallest num * 4, skip
                if remaining > nums[n-2] + nums[n-1]:
                    continue
                
                left = j + 1
                right = n - 1
                while left < right:
                    curr = nums[i] + nums[j] + nums[left] + nums[right]
                    if curr == target:
                        result.append([nums[i], nums[j], nums[left], nums[right]])
                        left += 1
                        right -= 1
                        # only do once for every left value
                        while left < right and nums[left] == nums[left - 1]:
                            left += 1
                        # only do once for every right value
                        while left < right and nums[right] == nums[right + 1]:
                            right -= 1
                    elif curr < target:
                        left += 1
                    else:
                        right -= 1
        
        return result
```
