## Idea
Set a smallest num, and then start to search to its right. If it is sum to 0, then append to the result list. Otherwise, if `add_num` > 0, then `k` moves inward; if `add_num` < 0, then `j` moves outward. Stop when `j < k`.

## Code
My solution:
```python
class Solution(object):
    def threeSum(self, nums):
        """
        :type nums: List[int]
        :rtype: List[List[int]]
        """
        result = []
        new_nums = sorted(nums)
        for i in range(0, len(nums)-1):
            j = i+1
            k = len(nums)-1
            curr = new_nums[i]
            num_sum = curr + new_nums[j] + new_nums[k]
            while j < k:
                if num_sum == 0:
                    add_num = sorted([new_nums[i], new_nums[j], new_nums[k]])
                    if add_num not in result:
                        result.append(add_num)
                    k -= 1
                    j += 1
                elif num_sum > 0:
                    k -= 1
                else:
                    j += 1
                num_sum = curr + new_nums[j] + new_nums[k]
        return result
```

## Ideal Solution
Same strategy, but use `set` instead.

```python
class Solution:
    def threeSum(self, nums: List[int]) -> List[List[int]]:
        target = 0
        nums.sort()
        s = set()
        output = []
        for i in range(len(nums)):
            if nums[i] > 0:
                break
            if i > 0 and nums[i] == nums[i-1]:
                continue
            j = i + 1
            k = len(nums) - 1
            while j < k:
                sum = nums[i] + nums[j] + nums[k]
                if sum == target:
                    s.add((nums[i], nums[j], nums[k]))
                    j += 1
                    k -= 1
                elif sum < target:
                    j += 1
                else:
                    k -= 1
        output = list(s)
        return output
```
